<p>主要编译过程：
LIB_DIRS = /arm-2012.03/lib/gcc/arm-none-linux-gnueabi/4.6.3/
CFLAGS  = -mcpu=arm926ej-s -mtune=arm926ej-s -g -Os -nostdlib -fno-builtin -Wall 
LDFLAGS = -nostdlib -format elf32-littlearm -T $(TOPDIR)/startup.lds -L $(LIB_DIRS)
AS = $(CROSS_COMPILE)as $(AFLAGS)
LD = $(CROSS_COMPILE)ld $(LDFLAGS)
CC = $(CROSS_COMPILE)gcc $(CFLAGS)
STRIP = $(CROSS_COMPILE)strip -R .note -R .comment -S
"$(LOADER).bin" :   $(OBJS)
      @echo assembly/compile done. linking.
      @$(LD) -o "$(LOADER)-elf32" $(OBJECTS) $(LIBS) -Map "$(LOADER).map"
      @echo linking done. strip.
      @$(STRIP) "$(LOADER)-elf32" -o "$(LOADER)-elf32-stripped"
      @echo strip done. objcopy.
      @$(OBJCOPY) "$(LOADER)-elf32" "$(LOADER).bin"
      @echo Build Successful!</p>
